import Base

# 2^d via fork/join recursion: each half of the split runs as its own
# task ("parallel let"), joined by the addition below. This is Bend's
# headline feature -- ordinary-looking recursive code that the compiler
# schedules across cores (or the GPU, with `!`) without you writing any
# thread/task code yourself.
def pow2(+d: Nat) -> U32:
  match d:
    case 0n:
      1
    case 1n+p:
      a b = pow2(p) pow2(p)
      (a + b : U32)

def main() -> IO(Unit):
  do IO<Unit>:
    IO.print(U32.show(pow2(10n)))

arrays: literal, write, read-as-pair
============================================================

import Base

# Array<T> is affine: at most one live owner, and a read hands back
# the array *beside* the element. Destructuring a computed expression
# directly is refused by the checker ("a match cannot scrutinize a
# computed value") -- thread it into a helper whose PARAMETER is that
# pair instead, same as this file's `cell`.
def cell(a: Array<U32> & U32) -> U32:
  (arr, v) = a
  v

# `a[i] <- v` on its own line implicitly rebinds `a` -- but only in a
# plain statement sequence like this one, not inside a `do` block
# (there, every line is a typed bind/let/step; build the array here
# and hand `main` only the finished value).
def build() -> U32:
  arr = [0 : U32*8n]  # 8 slots (power of two), all zero
  arr[0] <- 10
  arr[1] <- 32
  cell(arr[1])

def main() -> IO(Unit):
  do IO<Unit>:
    IO.print(U32.show(build()))

============================================================
law, proof by match, rewrite step
============================================================

import Base

# Bend's headline feature: state an intent as a `law` (a claim, which
# is itself just a type), then write a `def` of the same name that
# proves it. The checker rejects the file if the proof doesn't
# actually establish the claim -- this is what "blocks AI mistakes
# via proof" means in practice, not marketing copy.

law add_zero:
  for x: Nat
  {Nat.add(x, 0n) == x : Nat}

def add_zero(x):              # unannotated: fills the law, inherits its type
  match x:
    case 0n:
      {==}                     # base case: Nat.add(0n, 0n) reduces to 0n by computation
    case 1n+p:
      %add_zero(p) : {1n+Nat.add(p, 0n) == 1n+_ : Nat}   # rewrite using the inductive hypothesis
      {==}

def main() -> IO(Unit):
  do IO<Unit>:
    IO.print("add_zero proved for all Nat -- checked, not asserted")

import Base

# Template: `~f` is compile-time substituted and may be instantiated more
# than once; `apply_twice` calls it twice on the same argument.
def apply_twice(~f: U32 -> U32, x: U32) -> U32:
  f(f(x))

def inc(x: U32) -> U32:
  (x + 1 : U32)

@unsafe def spin(n: Nat) -> Nat:
  match n:
    case 0n:
      0n
    case 1n+p:
      spin(1n+p)

def swap(pair: U32 & U32) -> U32 & U32:
  (a, b) = pair
  (b, a)

def render(pair: U32 & U32) -> String:
  (x, y) = pair
  U32.show(x) ++ U32.show(y)

def show_swapped(pair: U32 & U32) -> String:
  render(swap(pair))

def main() -> IO(Unit):
  do IO<Unit>:
    IO.print(U32.show(apply_twice(inc, 5)))
    IO.print(show_swapped((1, 2)))
